CREATE TABLE refresh_tokens (
    id           SERIAL PRIMARY KEY,
    user_id      INTEGER NOT NULL,
    user_type    VARCHAR(10) NOT NULL CHECK (user_type IN ('student', 'admin')),
    token_hash   VARCHAR(255) NOT NULL,
    created_at   TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    expires_at   TIMESTAMP WITH TIME ZONE NOT NULL,
    is_revoked   BOOLEAN DEFAULT FALSE,
    ip_address   VARCHAR(45),
    device_info  VARCHAR(255)
);

-- Index: fast lookup by user (for listing sessions / revoking all)
CREATE INDEX idx_refresh_tokens_user 
    ON refresh_tokens (user_type, user_id);

-- Index: fast lookup by hash (for token validation)
CREATE INDEX idx_refresh_tokens_hash 
    ON refresh_tokens (token_hash) 
    WHERE is_revoked = FALSE;

-- Index: cleanup of expired tokens
CREATE INDEX idx_refresh_tokens_expires 
    ON refresh_tokens (expires_at) 
    WHERE is_revoked = FALSE;

CREATE TABLE clans (
    clan_id SERIAL PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    description TEXT,

    faculty_id INTEGER UNIQUE,
    created_by INTEGER,

    status VARCHAR(20) DEFAULT 'نشط',

    min_members INTEGER DEFAULT 50,

    created_at TIMESTAMP,
    updated_at TIMESTAMP,

    CONSTRAINT fk_clan_faculty
        FOREIGN KEY (faculty_id)
        REFERENCES faculties(faculty_id)
        ON DELETE CASCADE,

    CONSTRAINT fk_clan_admin
        FOREIGN KEY (created_by)
        REFERENCES admins(admin_id)
        ON DELETE SET NULL
);

CREATE TABLE clan_groups (
    group_id SERIAL PRIMARY KEY,
    name VARCHAR(100) NOT NULL,

    clan_id INTEGER NOT NULL,

    display_order INTEGER DEFAULT 1,

    created_at TIMESTAMP,
    updated_at TIMESTAMP,

    CONSTRAINT fk_group_clan
        FOREIGN KEY (clan_id)
        REFERENCES clans(clan_id)
        ON DELETE CASCADE
);


CREATE TABLE scout_members (
    scout_member_id SERIAL PRIMARY KEY,

    student_id INTEGER NOT NULL,
    clan_id INTEGER NOT NULL,
    group_id INTEGER,

    role VARCHAR(30) DEFAULT 'MEMBER',

    status VARCHAR(20) DEFAULT 'منتظر',

    reviewed_by INTEGER,

    rejection_reason TEXT,
    reviewed_at TIMESTAMP,
    joined_at TIMESTAMP,

    created_at TIMESTAMP,
    updated_at TIMESTAMP,

    CONSTRAINT fk_member_student
        FOREIGN KEY (student_id)
        REFERENCES students(student_id)
        ON DELETE CASCADE,

    CONSTRAINT fk_member_clan
        FOREIGN KEY (clan_id)
        REFERENCES clans(clan_id)
        ON DELETE CASCADE,

    CONSTRAINT fk_member_group
        FOREIGN KEY (group_id)
        REFERENCES clan_groups(group_id)
        ON DELETE SET NULL,

    CONSTRAINT fk_member_admin
        FOREIGN KEY (reviewed_by)
        REFERENCES admins(admin_id)
        ON DELETE SET NULL,

    CONSTRAINT unique_student_clan
        UNIQUE (student_id, clan_id)
);
